//@version=6
//=====================================================================
// HOW TO READ THE SIGNALS
//=====================================================================
//
// Each signal fires purely from price location relative to the 4-year
// AVWAP zones - there is no trend or confluence filter gating them
// anymore. A signal appears when price dips into a zone below the
// fair-value band and closes back above it (or, if reclaim
// confirmation is turned off, the moment price simply trades inside
// the zone). Think of the four signals as a ladder of discount depth,
// not a ranking of reliability:
//
//   BUY      - price briefly dipped below the fair-value band itself
//              and reclaimed it. The most common signal - a shallow
//              dip.
//
//   BUY+     - a real pullback (Zone 1, ~0.5x weekly ATR below fair
//              value) that got bought back up. Moderate discount.
//
//   BUY++    - a stronger pullback (Zone 2, ~1x ATR below). Less
//              frequent, bigger discount.
//
//   BUY+++   - a deep, panic-level flush (Zone 3, ~1.5x ATR below)
//              that got reclaimed. Rare, and historically the best
//              long-term entry price - but also the scariest to buy
//              into, since it often happens right when sentiment is
//              worst.
//
// More "+" symbols means a deeper discount, not a more reliable
// signal - accumulation here is purely a function of how far price
// has fallen toward (or into) a zone, independent of trend.
//
// The 40-week MA trend, composite trend, and AVWAP confluence count
// are still shown in the status table for context, but as of this
// version they no longer block or filter any signal.
//
//=====================================================================

//=====================================================================
// Identify latest four calendar years
//=====================================================================

int latestChartYear = year(last_bar_time)

int currentYear = latestChartYear
int previousYear = latestChartYear - 1
int twoYearsAgo = latestChartYear - 2
int threeYearsAgo = latestChartYear - 3

//=====================================================================
// Cumulative price-volume calculations
//=====================================================================

float barVolume = nz(volume, 0.0)

float cumulativePriceVolume = ta.cum(src * barVolume)
float cumulativeVolume = ta.cum(barVolume)

float previousCumulativePriceVolume =
     nz(cumulativePriceVolume[1], 0.0)

float previousCumulativeVolume =
     nz(cumulativeVolume[1], 0.0)

//=====================================================================
// Track each calendar year's running low (causal, no future leakage)
//=====================================================================

// This keeps a running lowest-low for whichever calendar year each
// bar belongs to, updating only with information available at that
// bar - no request.security lookahead is used anywhere in this
// script. For completed years the running low freezes at the year's
// true low once the year ends. For the in-progress year, the anchor
// can still move forward live as new lows print - that is expected
// behavior (the AVWAP "follows" the year's lowest point so far), not
// repainting, since no future data is ever used.

var float yearRunningLow = na

bool isNewCalendarYear = ta.change(year) != 0

if isNewCalendarYear or na(yearRunningLow)
    yearRunningLow := low
else
    yearRunningLow := math.min(yearRunningLow, low)

bool isRunningYearLow = low <= yearRunningLow

//=====================================================================
// Locate each year's lowest bar
//=====================================================================

bool anchorCurrentYear =
     year == currentYear and isRunningYearLow

bool anchorPreviousYear =
     year == previousYear and isRunningYearLow

bool anchorTwoYearsAgo =
     year == twoYearsAgo and isRunningYearLow

bool anchorThreeYearsAgo =
     year == threeYearsAgo and isRunningYearLow

//=====================================================================
// Store cumulative price-volume before each anchor
//=====================================================================

float anchorPriceVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativePriceVolume,
     0
)

//=====================================================================
// Store cumulative volume before each anchor
//=====================================================================

float anchorVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativeVolume,
     0
)

float anchorVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativeVolume,
     0
)

float anchorVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativeVolume,
     0
)

float anchorVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativeVolume,
     0
)

//=====================================================================
// Store anchor times
//=====================================================================

int anchorTimeCurrent = ta.valuewhen(
     anchorCurrentYear,
     time,
     0
)

int anchorTimePrevious = ta.valuewhen(
     anchorPreviousYear,
     time,
     0
)

int anchorTimeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     time,
     0
)

int anchorTimeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     time,
     0
)

//=====================================================================
// AVWAP function
//=====================================================================

calculateAVWAP(
     float storedPriceVolume,
     float storedVolume,
     int storedAnchorTime
) =>
    float volumeFromAnchor = cumulativeVolume - storedVolume

    bool validAnchor =
         not na(storedPriceVolume) and
         not na(storedVolume) and
         not na(storedAnchorTime) and
         time >= storedAnchorTime and
         volumeFromAnchor > 0.0

    validAnchor
         ? (cumulativePriceVolume - storedPriceVolume) / volumeFromAnchor
         : na

//=====================================================================
// Calculate individual AVWAPs
//=====================================================================

float avwapCurrentYear = calculateAVWAP(
     anchorPriceVolumeCurrent,
     anchorVolumeCurrent,
     anchorTimeCurrent
)

float avwapPreviousYear = calculateAVWAP(
     anchorPriceVolumePrevious,
     anchorVolumePrevious,
     anchorTimePrevious
)

float avwapTwoYearsAgo = calculateAVWAP(
     anchorPriceVolumeTwoYears,
     anchorVolumeTwoYears,
     anchorTimeTwoYears
)

float avwapThreeYearsAgo = calculateAVWAP(
     anchorPriceVolumeThreeYears,
     anchorVolumeThreeYears,
     anchorTimeThreeYears
)

//=====================================================================
// Recency-weighted composite
//=====================================================================

float activeWeightCurrent =
     na(avwapCurrentYear) ? 0.0 : weightCurrentYear

float activeWeightPrevious =
     na(avwapPreviousYear) ? 0.0 : weightPreviousYear

float activeWeightTwoYears =
     na(avwapTwoYearsAgo) ? 0.0 : weightTwoYearsAgo

float activeWeightThreeYears =
     na(avwapThreeYearsAgo) ? 0.0 : weightThreeYearsAgo

float activeValueCurrent =
     na(avwapCurrentYear)
         ? 0.0
         : avwapCurrentYear * weightCurrentYear

float activeValuePrevious =
     na(avwapPreviousYear)
         ? 0.0
         : avwapPreviousYear * weightPreviousYear

float activeValueTwoYears =
     na(avwapTwoYearsAgo)
         ? 0.0
         : avwapTwoYearsAgo * weightTwoYearsAgo

float activeValueThreeYears =
     na(avwapThreeYearsAgo)
         ? 0.0
         : avwapThreeYearsAgo * weightThreeYearsAgo

float activeTotalWeight =
     activeWeightCurrent +
     activeWeightPrevious +
     activeWeightTwoYears +
     activeWeightThreeYears

float activeWeightedSum =
     activeValueCurrent +
     activeValuePrevious +
     activeValueTwoYears +
     activeValueThreeYears

float weightedAverageAVWAP =
     activeTotalWeight > 0.0
         ? activeWeightedSum / activeTotalWeight
         : na

//=====================================================================
// Equal-weight composite
//=====================================================================

float equalValueCurrent =
     na(avwapCurrentYear) ? 0.0 : avwapCurrentYear

float equalValuePrevious =
     na(avwapPreviousYear) ? 0.0 : avwapPreviousYear

float equalValueTwoYears =
     na(avwapTwoYearsAgo) ? 0.0 : avwapTwoYearsAgo

float equalValueThreeYears =
     na(avwapThreeYearsAgo) ? 0.0 : avwapThreeYearsAgo

float availableAVWAPCount =
     (na(avwapCurrentYear) ? 0.0 : 1.0) +
     (na(avwapPreviousYear) ? 0.0 : 1.0) +
     (na(avwapTwoYearsAgo) ? 0.0 : 1.0) +
     (na(avwapThreeYearsAgo) ? 0.0 : 1.0)

float equalWeightAVWAP =
     availableAVWAPCount > 0.0
         ? (
             equalValueCurrent +
             equalValuePrevious +
             equalValueTwoYears +
             equalValueThreeYears
           ) / availableAVWAPCount
         : na

//=====================================================================
// Central accumulation band
//=====================================================================

// The region between the recency-weighted composite and equal-weight
// composite is treated as the central "fair value" band. Price above
// this band is trading rich relative to 4 years of accumulated buying;
// price below it is where buy zones 1-3 are measured from.

float centralBandUpper =
     math.max(weightedAverageAVWAP, equalWeightAVWAP)

float centralBandLower =
     math.min(weightedAverageAVWAP, equalWeightAVWAP)

float centralBandMidpoint =
     (centralBandUpper + centralBandLower) / 2.0

//=====================================================================
// Weekly ATR and weekly moving average
//=====================================================================

float weeklyATR = request.security(
     syminfo.tickerid,
     "W",
     ta.atr(weeklyATRLength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

float weeklyMA = request.security(
     syminfo.tickerid,
     "W",
     ta.sma(close, weeklyMALength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

float weeklyMAPrevious = request.security(
     syminfo.tickerid,
     "W",
     ta.sma(close, weeklyMALength)[1],
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

//=====================================================================
// ATR-based accumulation ("buy") zones
//=====================================================================

// Zones are measured underneath the lower edge of the central band.
// Zone 1 = mild pullback, Zone 2 = strong pullback, Zone 3 = deep /
// panic-level pullback. Deeper zones are rarer and historically offer
// a bigger long-term discount, at the cost of being harder to time.

float initialAccumulationZone =
     centralBandLower - weeklyATR * initialZoneATR

float strongAccumulationZone =
     centralBandLower - weeklyATR * strongZoneATR

float deepAccumulationZone =
     centralBandLower - weeklyATR * deepZoneATR

//=====================================================================
// Composite and moving-average slopes (display context only)
//=====================================================================

float weeklyComposite = request.security(
     syminfo.tickerid,
     "W",
     weightedAverageAVWAP,
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

float compositeSlope =
     weeklyComposite - weeklyComposite[slopeLookbackWeeks]

bool compositeRising = compositeSlope > 0.0
bool compositeFalling = compositeSlope < 0.0

bool weeklyMARising =
     not na(weeklyMAPrevious) and weeklyMA >= weeklyMAPrevious

color mainCompositeColor =
     not colorCompositeBySlope
         ? colorFlatComposite
         : compositeRising
             ? colorRisingComposite
             : compositeFalling
                 ? colorFallingComposite
                 : colorFlatComposite

//=====================================================================
// AVWAP confluence count (display context only)
//=====================================================================

float confluenceDistance =
     weeklyATR * confluenceDistanceATR

bool currentAVWAPNearby =
     not na(avwapCurrentYear) and
     math.abs(close - avwapCurrentYear) <= confluenceDistance

bool previousAVWAPNearby =
     not na(avwapPreviousYear) and
     math.abs(close - avwapPreviousYear) <= confluenceDistance

bool twoYearAVWAPNearby =
     not na(avwapTwoYearsAgo) and
     math.abs(close - avwapTwoYearsAgo) <= confluenceDistance

bool threeYearAVWAPNearby =
     not na(avwapThreeYearsAgo) and
     math.abs(close - avwapThreeYearsAgo) <= confluenceDistance

int confluenceCount =
     (currentAVWAPNearby ? 1 : 0) +
     (previousAVWAPNearby ? 1 : 0) +
     (twoYearAVWAPNearby ? 1 : 0) +
     (threeYearAVWAPNearby ? 1 : 0)

//=====================================================================
// Buy signals - driven purely by price location relative to zones.
// No trend, composite, or confluence condition is required anymore.
//=====================================================================

// If reclaim confirmation is selected, the signal appears when price
// crosses back above the relevant zone (buyers have stepped back in).
//
// Otherwise, the signal appears the moment price trades inside or
// below the zone (earlier, but less confirmed).

bool normalAccumulationSignal =
     confirmationOnReclaim
         ? ta.crossover(close, centralBandLower)
         : close <= centralBandLower and close > initialAccumulationZone

bool initialAccumulationSignal =
     confirmationOnReclaim
         ? ta.crossover(close, initialAccumulationZone)
         : close <= initialAccumulationZone and
           close > strongAccumulationZone

bool strongAccumulationSignal =
     confirmationOnReclaim
         ? ta.crossover(close, strongAccumulationZone)
         : close <= strongAccumulationZone and
           close > deepAccumulationZone

bool deepAccumulationSignal =
     confirmationOnReclaim
         ? ta.crossover(close, deepAccumulationZone)
         : close <= deepAccumulationZone

bool showNormalSignal =
     showAccumulationSignals and normalAccumulationSignal

bool showInitialSignal =
     showAccumulationSignals and initialAccumulationSignal

bool showStrongSignal =
     showAccumulationSignals and strongAccumulationSignal

bool showDeepSignal =
     showAccumulationSignals and deepAccumulationSignal
