// Identify latest four calendar years
//=====================================================================

int latestChartYear = year(last_bar_time)

int threeYearsAgo = latestChartYear - 3

//=====================================================================
// Cumulative price-volume calculations
//=====================================================================

float barVolume = nz(volume, 0.0)

float cumulativePriceVolume = ta.cum(src * barVolume)
float cumulativeVolume = ta.cum(barVolume)

float previousCumulativePriceVolume =
     nz(cumulativePriceVolume[1], 0.0)

float previousCumulativeVolume =
     nz(cumulativeVolume[1], 0.0)

//=====================================================================
// Track each calendar year's running low
//=====================================================================
//
// No future annual values or lookahead are used.
//
// During an active calendar year, the anchor moves forward when a
// strictly lower price is recorded.
//
//=====================================================================

var float yearRunningLow = na

bool isNewCalendarYear =
     barstate.isfirst or year != year[1]

bool isRunningYearLow =
     isNewCalendarYear or
     na(yearRunningLow) or
     low < yearRunningLow

if isNewCalendarYear or na(yearRunningLow)
    yearRunningLow := low
else if low < yearRunningLow
    yearRunningLow := low

//=====================================================================
// Identify the anchor events for each year
//=====================================================================

bool anchorCurrentYear =
     year == currentYear and isRunningYearLow

bool anchorPreviousYear =
     year == previousYear and isRunningYearLow

bool anchorTwoYearsAgo =
     year == twoYearsAgo and isRunningYearLow

bool anchorThreeYearsAgo =
     year == threeYearsAgo and isRunningYearLow

//=====================================================================
// Store cumulative price-volume before each anchor
//=====================================================================

float anchorPriceVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativePriceVolume,
     0
)

//=====================================================================
// Store cumulative volume before each anchor
//=====================================================================

float anchorVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativeVolume,
     0
)

float anchorVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativeVolume,
     0
)

float anchorVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativeVolume,
     0
)

float anchorVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativeVolume,
     0
)

//=====================================================================
// Store anchor times
//=====================================================================

int anchorTimeCurrent = ta.valuewhen(
     anchorCurrentYear,
     time,
     0
)

int anchorTimePrevious = ta.valuewhen(
     anchorPreviousYear,
     time,
     0
)

int anchorTimeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     time,
     0
)

int anchorTimeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     time,
     0
)

//=====================================================================
// AVWAP function
//=====================================================================

calculateAVWAP(
     float storedPriceVolume,
     float storedVolume,
     int storedAnchorTime
) =>
    float volumeFromAnchor =
         cumulativeVolume - storedVolume

    bool validAnchor =
         not na(storedPriceVolume) and
         not na(storedVolume) and
         not na(storedAnchorTime) and
         time >= storedAnchorTime and
         volumeFromAnchor > 0.0

    validAnchor
         ? (cumulativePriceVolume - storedPriceVolume) /
           volumeFromAnchor
         : na

//=====================================================================
// Calculate individual AVWAPs
//=====================================================================

float avwapCurrentYear = calculateAVWAP(
     anchorPriceVolumeCurrent,
     anchorVolumeCurrent,
     anchorTimeCurrent
)

float avwapPreviousYear = calculateAVWAP(
     anchorPriceVolumePrevious,
     anchorVolumePrevious,
     anchorTimePrevious
)

float avwapTwoYearsAgo = calculateAVWAP(
     anchorPriceVolumeTwoYears,
     anchorVolumeTwoYears,
     anchorTimeTwoYears
)

float avwapThreeYearsAgo = calculateAVWAP(
     anchorPriceVolumeThreeYears,
     anchorVolumeThreeYears,
     anchorTimeThreeYears
)

//=====================================================================
// Recency-weighted composite
//=====================================================================

float activeWeightCurrent =
     na(avwapCurrentYear)
         ? 0.0
         : weightCurrentYear

float activeWeightPrevious =
     na(avwapPreviousYear)
         ? 0.0
         : weightPreviousYear

float activeWeightTwoYears =
     na(avwapTwoYearsAgo)
         ? 0.0
         : weightTwoYearsAgo

float activeWeightThreeYears =
     na(avwapThreeYearsAgo)
         ? 0.0
         : weightThreeYearsAgo

float activeValueCurrent =
     na(avwapCurrentYear)
         ? 0.0
         : avwapCurrentYear * weightCurrentYear

float activeValuePrevious =
     na(avwapPreviousYear)
         ? 0.0
         : avwapPreviousYear * weightPreviousYear

float activeValueTwoYears =
     na(avwapTwoYearsAgo)
         ? 0.0
         : avwapTwoYearsAgo * weightTwoYearsAgo

float activeValueThreeYears =
     na(avwapThreeYearsAgo)
         ? 0.0
         : avwapThreeYearsAgo * weightThreeYearsAgo

float activeTotalWeight =
     activeWeightCurrent +
     activeWeightPrevious +
     activeWeightTwoYears +
     activeWeightThreeYears

float activeWeightedSum =
     activeValueCurrent +
     activeValuePrevious +
     activeValueTwoYears +
     activeValueThreeYears

float weightedAverageAVWAP =
     activeTotalWeight > 0.0
         ? activeWeightedSum / activeTotalWeight
         : na

//=====================================================================
// Equal-weight composite
//=====================================================================

float equalValueCurrent =
     na(avwapCurrentYear)
         ? 0.0
         : avwapCurrentYear

float equalValuePrevious =
     na(avwapPreviousYear)
         ? 0.0
         : avwapPreviousYear

float equalValueTwoYears =
     na(avwapTwoYearsAgo)
         ? 0.0
         : avwapTwoYearsAgo

float equalValueThreeYears =
     na(avwapThreeYearsAgo)
         ? 0.0
         : avwapThreeYearsAgo

float availableAVWAPCount =
     (na(avwapCurrentYear) ? 0.0 : 1.0) +
     (na(avwapPreviousYear) ? 0.0 : 1.0) +
     (na(avwapTwoYearsAgo) ? 0.0 : 1.0) +
     (na(avwapThreeYearsAgo) ? 0.0 : 1.0)

float equalWeightAVWAP =
     availableAVWAPCount > 0.0
         ? (
             equalValueCurrent +
             equalValuePrevious +
             equalValueTwoYears +
             equalValueThreeYears
           ) / availableAVWAPCount
         : na

//=====================================================================
// Central fair-value band
//=====================================================================
//
// The technical fair-value band is the area between:
//
// 1. The recency-weighted AVWAP composite
// 2. The equal-weight AVWAP composite
//
//=====================================================================

float centralBandUpper =
     not na(weightedAverageAVWAP) and
     not na(equalWeightAVWAP)
         ? math.max(
             weightedAverageAVWAP,
             equalWeightAVWAP
           )
         : na

float centralBandLower =
     not na(weightedAverageAVWAP) and
     not na(equalWeightAVWAP)
         ? math.min(
             weightedAverageAVWAP,
             equalWeightAVWAP
           )
         : na

float centralBandMidpoint =
     not na(centralBandUpper) and
     not na(centralBandLower)
         ? (centralBandUpper + centralBandLower) / 2.0
         : na

//=====================================================================
// Weekly ATR and moving average
//=====================================================================

float weeklyATR = request.security(
     syminfo.tickerid,
     "W",
     ta.atr(weeklyATRLength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

float weeklyMA = request.security(
     syminfo.tickerid,
     "W",
     ta.sma(close, weeklyMALength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

//=====================================================================
// Discount zones below fair value
//=====================================================================

float initialAccumulationZone =
     not na(centralBandLower) and
     not na(weeklyATR)
         ? centralBandLower -
           weeklyATR * initialZoneATR
         : na

float strongAccumulationZone =
     not na(centralBandLower) and
     not na(weeklyATR)
         ? centralBandLower -
           weeklyATR * strongZoneATR
         : na

float deepAccumulationZone =
     not na(centralBandLower) and
     not na(weeklyATR)
         ? centralBandLower -
           weeklyATR * deepZoneATR
         : na

//=====================================================================
// Fair-value position
//=====================================================================
//
// -1.0 = Far above fair value
//  0.0 = Inside fair-value band
// +1.0 = At or below Zone 3
//
// Above and below fair value are measured independently using weekly
// ATR so the score adjusts to the volatility of each stock.
//
//=====================================================================

float aboveFairValueDistance =
     weeklyATR * expensiveZoneATR

float belowFairValueDistance =
     weeklyATR * deepZoneATR

float distanceAboveFairValue =
     not na(centralBandUpper)
         ? math.max(
             0.0,
             close - centralBandUpper
           )
         : 0.0

float distanceBelowFairValue =
     not na(centralBandLower)
         ? math.max(
             0.0,
             centralBandLower - close
           )
         : 0.0

float aboveFairValueScore =
     not na(weeklyATR) and
     aboveFairValueDistance > 0.0
         ? math.min(
             1.0,
             distanceAboveFairValue /
             aboveFairValueDistance
           )
         : 0.0

float belowFairValueScore =
     not na(weeklyATR) and
     belowFairValueDistance > 0.0
         ? math.min(
             1.0,
             distanceBelowFairValue /
             belowFairValueDistance
           )
         : 0.0

float fairValuePosition =
     na(centralBandUpper) or
     na(centralBandLower)
         ? 0.0
         : close > centralBandUpper
             ? -aboveFairValueScore
             : close < centralBandLower
                 ? belowFairValueScore
                 : 0.0

//=====================================================================
// SIP allocation
//=====================================================================
//
// Above fair value:
//
//   Allocation moves from normal SIP toward minimum SIP.
//
// Inside fair value:
//
//   Allocation remains at normal SIP.
//
// Below fair value:
//
//   Allocation moves from normal SIP toward maximum SIP.
//
//=====================================================================

float validatedMinSipPct =
     math.min(minSipPct, normalSipPct)

float validatedNormalSipPct =
     math.max(
         validatedMinSipPct,
         math.min(normalSipPct, maxSipPct)
     )

float validatedMaxSipPct =
     math.max(
         maxSipPct,
         validatedNormalSipPct
     )

float recommendedSipPct =
     fairValuePosition < 0.0
         ? validatedNormalSipPct -
           math.abs(fairValuePosition) *
           (
               validatedNormalSipPct -
               validatedMinSipPct
           )
         : fairValuePosition > 0.0
             ? validatedNormalSipPct +
               fairValuePosition *
               (
                   validatedMaxSipPct -
                   validatedNormalSipPct
               )
             : validatedNormalSipPct

recommendedSipPct :=
     math.max(
         validatedMinSipPct,
         math.min(
             validatedMaxSipPct,
             recommendedSipPct
         )
     )

     na(centralBandLower)
         ? "Insufficient Data"
