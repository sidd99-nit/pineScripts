//=====================================================================
// HOW TO READ THIS CHART
//=====================================================================
//
// This is a pure "value for money" gauge for monthly SIP investing -
// designed for fundamentally strong stocks / ETFs.
//
//   - The two thick lines (and the band between them) mark the
//     4-year AVWAP fair-value zone, built from where price bottomed
//     in each of the last 4 calendar years.
//   - The three thinner lines below the band are reference distances
//     (in weekly ATR) - Zone 1 (mild), Zone 2 (strong), Zone 3
//     (deep) discounts below fair value.
//   - The background tint is the actual "value for money" read:
//     it shades toward green the further price sits below fair
//     value, and stays neutral/untinted at or above fair value.
//     Darker green = a better price to buy at this month.
//
// SIP ALLOCATION (shown in the table):
//   Recommended % of your planned SIP budget for this ticker =
//       55% + (Value Score Ã 45%)
//   â ranges from 55% (expensive) to 100% (deep discount)
//   Never exceeds 100% of the planned budget.
//
// Use it on your SIP date: look at the table for the exact %
// of budget to deploy into that strong stock/ETF this month.
//
//=====================================================================

//=====================================================================
// Identify latest four calendar years
//=====================================================================

int latestChartYear = year(last_bar_time)

int currentYear = latestChartYear
int previousYear = latestChartYear - 1
int twoYearsAgo = latestChartYear - 2
int threeYearsAgo = latestChartYear - 3

//=====================================================================
// Cumulative price-volume calculations
//=====================================================================

float barVolume = nz(volume, 0.0)

float cumulativePriceVolume = ta.cum(src * barVolume)
float cumulativeVolume = ta.cum(barVolume)

float previousCumulativePriceVolume =
     nz(cumulativePriceVolume[1], 0.0)

float previousCumulativeVolume =
     nz(cumulativeVolume[1], 0.0)

//=====================================================================
// Track each calendar year's running low (causal, no future leakage)
//=====================================================================

// Keeps a running lowest-low for whichever calendar year each bar
// belongs to, updating only with information available at that bar -
// no request.security lookahead is used anywhere in this script. For
// completed years the running low freezes at the year's true low
// once the year ends. For the in-progress year, the anchor can still
// move forward live as new lows print - that is expected, not
// repainting, since no future data is ever used.

var float yearRunningLow = na

bool isNewCalendarYear = ta.change(year) != 0

if isNewCalendarYear or na(yearRunningLow)
    yearRunningLow := low
else
    yearRunningLow := math.min(yearRunningLow, low)

bool isRunningYearLow = low <= yearRunningLow

//=====================================================================
// Locate each year's lowest bar
//=====================================================================

bool anchorCurrentYear =
     year == currentYear and isRunningYearLow

bool anchorPreviousYear =
     year == previousYear and isRunningYearLow

bool anchorTwoYearsAgo =
     year == twoYearsAgo and isRunningYearLow

bool anchorThreeYearsAgo =
     year == threeYearsAgo and isRunningYearLow

//=====================================================================
// Store cumulative price-volume before each anchor
//=====================================================================

float anchorPriceVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativePriceVolume,
     0
)

float anchorPriceVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativePriceVolume,
     0
)

//=====================================================================
// Store cumulative volume before each anchor
//=====================================================================

float anchorVolumeCurrent = ta.valuewhen(
     anchorCurrentYear,
     previousCumulativeVolume,
     0
)

float anchorVolumePrevious = ta.valuewhen(
     anchorPreviousYear,
     previousCumulativeVolume,
     0
)

float anchorVolumeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     previousCumulativeVolume,
     0
)

float anchorVolumeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     previousCumulativeVolume,
     0
)

//=====================================================================
// Store anchor times
//=====================================================================

int anchorTimeCurrent = ta.valuewhen(
     anchorCurrentYear,
     time,
     0
)

int anchorTimePrevious = ta.valuewhen(
     anchorPreviousYear,
     time,
     0
)

int anchorTimeTwoYears = ta.valuewhen(
     anchorTwoYearsAgo,
     time,
     0
)

int anchorTimeThreeYears = ta.valuewhen(
     anchorThreeYearsAgo,
     time,
     0
)

//=====================================================================
// AVWAP function
//=====================================================================

calculateAVWAP(
     float storedPriceVolume,
     float storedVolume,
     int storedAnchorTime
) =>
    float volumeFromAnchor = cumulativeVolume - storedVolume

    bool validAnchor =
         not na(storedPriceVolume) and
         not na(storedVolume) and
         not na(storedAnchorTime) and
         time >= storedAnchorTime and
         volumeFromAnchor > 0.0

    validAnchor
         ? (cumulativePriceVolume - storedPriceVolume) / volumeFromAnchor
         : na

//=====================================================================
// Calculate individual AVWAPs
//=====================================================================

float avwapCurrentYear = calculateAVWAP(
     anchorPriceVolumeCurrent,
     anchorVolumeCurrent,
     anchorTimeCurrent
)

float avwapPreviousYear = calculateAVWAP(
     anchorPriceVolumePrevious,
     anchorVolumePrevious,
     anchorTimePrevious
)

float avwapTwoYearsAgo = calculateAVWAP(
     anchorPriceVolumeTwoYears,
     anchorVolumeTwoYears,
     anchorTimeTwoYears
)

float avwapThreeYearsAgo = calculateAVWAP(
     anchorPriceVolumeThreeYears,
     anchorVolumeThreeYears,
     anchorTimeThreeYears
)

//=====================================================================
// Recency-weighted composite
//=====================================================================

float activeWeightCurrent =
     na(avwapCurrentYear) ? 0.0 : weightCurrentYear

float activeWeightPrevious =
     na(avwapPreviousYear) ? 0.0 : weightPreviousYear

float activeWeightTwoYears =
     na(avwapTwoYearsAgo) ? 0.0 : weightTwoYearsAgo

float activeWeightThreeYears =
     na(avwapThreeYearsAgo) ? 0.0 : weightThreeYearsAgo

float activeValueCurrent =
     na(avwapCurrentYear)
         ? 0.0
         : avwapCurrentYear * weightCurrentYear

float activeValuePrevious =
     na(avwapPreviousYear)
         ? 0.0
         : avwapPreviousYear * weightPreviousYear

float activeValueTwoYears =
     na(avwapTwoYearsAgo)
         ? 0.0
         : avwapTwoYearsAgo * weightTwoYearsAgo

float activeValueThreeYears =
     na(avwapThreeYearsAgo)
         ? 0.0
         : avwapThreeYearsAgo * weightThreeYearsAgo

float activeTotalWeight =
     activeWeightCurrent +
     activeWeightPrevious +
     activeWeightTwoYears +
     activeWeightThreeYears

float activeWeightedSum =
     activeValueCurrent +
     activeValuePrevious +
     activeValueTwoYears +
     activeValueThreeYears

float weightedAverageAVWAP =
     activeTotalWeight > 0.0
         ? activeWeightedSum / activeTotalWeight
         : na

//=====================================================================
// Equal-weight composite
//=====================================================================

float equalValueCurrent =
     na(avwapCurrentYear) ? 0.0 : avwapCurrentYear

float equalValuePrevious =
     na(avwapPreviousYear) ? 0.0 : avwapPreviousYear

float equalValueTwoYears =
     na(avwapTwoYearsAgo) ? 0.0 : avwapTwoYearsAgo

float equalValueThreeYears =
     na(avwapThreeYearsAgo) ? 0.0 : avwapThreeYearsAgo

float availableAVWAPCount =
     (na(avwapCurrentYear) ? 0.0 : 1.0) +
     (na(avwapPreviousYear) ? 0.0 : 1.0) +
     (na(avwapTwoYearsAgo) ? 0.0 : 1.0) +
     (na(avwapThreeYearsAgo) ? 0.0 : 1.0)

float equalWeightAVWAP =
     availableAVWAPCount > 0.0
         ? (
             equalValueCurrent +
             equalValuePrevious +
             equalValueTwoYears +
             equalValueThreeYears
           ) / availableAVWAPCount
         : na

//=====================================================================
// Central fair-value band
//=====================================================================

float centralBandUpper =
     math.max(weightedAverageAVWAP, equalWeightAVWAP)

float centralBandLower =
     math.min(weightedAverageAVWAP, equalWeightAVWAP)

//=====================================================================
// Weekly ATR and weekly moving average
//=====================================================================

float weeklyATR = request.security(
     syminfo.tickerid,
     "W",
     ta.atr(weeklyATRLength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

float weeklyMA = request.security(
     syminfo.tickerid,
     "W",
     ta.sma(close, weeklyMALength),
     gaps = barmerge.gaps_off,
     lookahead = barmerge.lookahead_off
)

//=====================================================================
// Reference discount-zone lines
//=====================================================================

float initialAccumulationZone =
     centralBandLower - weeklyATR * initialZoneATR

float strongAccumulationZone =
     centralBandLower - weeklyATR * strongZoneATR

float deepAccumulationZone =
     centralBandLower - weeklyATR * deepZoneATR

//=====================================================================
// Value-for-money score
//=====================================================================

// 0.0  = price at or above fair value (no discount)
// 1.0  = price has fallen exactly to Zone 3 depth (deepest discount)
// Values are clamped to [0, 1] purely for the background gradient;
// price can still fall further below Zone 3, it just keeps the
// strongest tint rather than getting darker still.

float rawDiscountDepth =
     (not na(weeklyATR) and weeklyATR > 0.0 and not na(centralBandLower))
         ? (centralBandLower - close) / (weeklyATR * deepZoneATR)
         : na

float valueScore =
     na(rawDiscountDepth)
         ? 0.0
         : math.max(0.0, math.min(1.0, rawDiscountDepth))

//=====================================================================
// SIP recommended percentage (capped at 100%)
//=====================================================================

// Linear: minSipPct + valueScore * (maxSipPct - minSipPct)
// Default: 55% + valueScore * 45%  â  55% â¦ 100%

float sipRange = math.max(0.0, maxSipPct - minSipPct)
float recommendedSipPct =
     math.min(100.0, minSipPct + valueScore * sipRange)

//=====================================================================

plot(

//=====================================================================
// SIP Allocation Table
//=====================================================================

var table sipTable = na

if showSipTable and barstate.islast
    // Determine table position
    string pos =
         tablePosition == "Top Left"     ? position.top_left :
         tablePosition == "Top Right"    ? position.top_right :
         tablePosition == "Bottom Left"  ? position.bottom_left :
                                           position.bottom_right

    if na(sipTable)
        sipTable := table.new(pos, 2, 4, bgcolor = color.new(color.black, 15), border_width = 1, border_color = color.gray)

    // Header
    table.cell(sipTable, 0, 0, "SIP VALUE GAUGE", text_color = color.white, text_size = size.small, bgcolor = color.new(color.teal, 30))
    table.cell(sipTable, 1, 0, "", bgcolor = color.new(color.teal, 30))

    // Value Score
    table.cell(sipTable, 0, 1, "Value Score", text_color = color.silver, text_size = size.normal)
    table.cell(sipTable, 1, 1, str.tostring(valueScore, "#.##"), text_color = color.white, text_size = size.normal)

    // Recommended %
    color pctColor =
         recommendedSipPct >= 95 ? color.lime :
         recommendedSipPct >= 80 ? color.green :
         recommendedSipPct >= 65 ? color.yellow :
                                   color.orange

    table.cell(sipTable, 0, 2, "Suggested SIP", text_color = color.silver, text_size = size.normal)
    table.cell(sipTable, 1, 2, str.tostring(recommendedSipPct, "#.#") + "%", text_color = pctColor, text_size = size.large)

    // Interpretation line
    string zoneText =
         valueScore >= 0.70 ? "Deep Discount â Full amount" :
         valueScore >= 0.45 ? "Strong Discount â High allocation" :
         valueScore >= 0.20 ? "Mild Discount â Moderate allocation" :
                              "Fair / Expensive â Reduced allocation"

    table.cell(sipTable, 0, 3, zoneText, text_color = color.gray, text_size = size.tiny, bgcolor = color.new(color.black, 40))
    table.merge_cells(sipTable, 0, 3, 1, 3)
